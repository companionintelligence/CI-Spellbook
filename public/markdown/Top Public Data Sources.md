## Data Ingestion Tips

- Use rate-limited, paginated API clients (e.g., Python `requests` with `backoff`).
- Cache responses locally with expiration (e.g., Redis, disk cache) to avoid rate limits.
- Validate and normalize schemas: unify date formats, handle missing fields, and standardize units.
- For large datasets, use streaming processing (Apache Kafka, AWS Kinesis) and columnar formats (Parquet, Delta Lake).
- Implement data quality checks and monitoring to detect schema changes and anomalies.

## API Best Practices

- Respect API terms: monitor usage against quotas and attribution requirements.
- Use exponential backoff on 429/5xx errors.
- Secure API keys: store in environment variables or secrets manager.
- Utilize pagination and filtering to reduce payload size and improve performance.
- Implement proper logging and monitoring for API health and usage patterns.
- Use webhook subscriptions when available to get real-time updates instead of polling.

### Python Data Ingestion Pipeline
```python
import requests
import pandas as pd
import time
import json
from datetime import datetime, timedelta

class DataIngestionPipeline:
    def __init__(self, api_key=None, rate_limit_delay=1):
        self.api_key = api_key
        self.rate_limit_delay = rate_limit_delay
        self.session = requests.Session()
        
    def fetch_with_retry(self, url, params=None, max_retries=3):
        """Fetch data with exponential backoff retry logic"""
        for attempt in range(max_retries):
            try:
                response = self.session.get(url, params=params, timeout=30)
                response.raise_for_status()
                time.sleep(self.rate_limit_delay)
                return response.json()
            except requests.exceptions.RequestException as e:
                if attempt == max_retries - 1:
                    raise e
                wait_time = (2 ** attempt) + 1
                print(f"Attempt {attempt + 1} failed, retrying in {wait_time}s...")
                time.sleep(wait_time)
    
    def fetch_world_bank_data(self, indicator, country='all', date_range='2000:2023'):
        """Fetch World Bank indicator data"""
        url = f"https://api.worldbank.org/v2/country/{country}/indicator/{indicator}"
        params = {
            'date': date_range,
            'format': 'json',
            'per_page': 1000
        }
        
        data = self.fetch_with_retry(url, params)
        if data and len(data) > 1:
            return pd.DataFrame(data[1])
        return pd.DataFrame()
    
    def fetch_openweather_data(self, city, api_key):
        """Fetch current weather data"""
        url = "https://api.openweathermap.org/data/2.5/weather"
        params = {
            'q': city,
            'appid': api_key,
            'units': 'metric'
        }
        
        return self.fetch_with_retry(url, params)

# Usage example
pipeline = DataIngestionPipeline(rate_limit_delay=0.5)

# Fetch GDP data from World Bank
gdp_data = pipeline.fetch_world_bank_data('NY.GDP.MKTP.CD')
print(f"Retrieved {len(gdp_data)} GDP records")

# Save to file
gdp_data.to_csv('world_bank_gdp.csv', index=False)
```

### Real-time Data Streaming Example
```python
import asyncio
import aiohttp
import json
from datetime import datetime

class RealTimeDataStreamer:
    def __init__(self, api_endpoints):
        self.api_endpoints = api_endpoints
        self.data_buffer = []
        
    async def fetch_endpoint(self, session, endpoint):
        """Fetch data from a single endpoint"""
        try:
            async with session.get(endpoint['url']) as response:
                data = await response.json()
                timestamp = datetime.utcnow().isoformat()
                
                return {
                    'source': endpoint['name'],
                    'timestamp': timestamp,
                    'data': data
                }
        except Exception as e:
            print(f"Error fetching from {endpoint['name']}: {e}")
            return None
    
    async def stream_data(self, interval=60):
        """Stream data from multiple endpoints"""
        async with aiohttp.ClientSession() as session:
            while True:
                tasks = [
                    self.fetch_endpoint(session, endpoint) 
                    for endpoint in self.api_endpoints
                ]
                
                results = await asyncio.gather(*tasks, return_exceptions=True)
                
                for result in results:
                    if result and not isinstance(result, Exception):
                        self.data_buffer.append(result)
                        print(f"Collected data from {result['source']}")
                
                # Process buffer when it reaches certain size
                if len(self.data_buffer) >= 10:
                    await self.process_batch()
                
                await asyncio.sleep(interval)
    
    async def process_batch(self):
        """Process accumulated data batch"""
        if self.data_buffer:
            # Save to file or database
            with open(f'streaming_data_{datetime.now().strftime("%Y%m%d_%H%M")}.json', 'w') as f:
                json.dump(self.data_buffer, f, indent=2)
            
            print(f"Processed batch of {len(self.data_buffer)} records")
            self.data_buffer.clear()

```python
import pandas as pd
import numpy as np
from typing import Dict, List, Any

class DataQualityValidator:
    def __init__(self):
        self.quality_report = {}
    
    def validate_completeness(self, df: pd.DataFrame) -> Dict[str, float]:
        """Check for missing values in each column"""
        missing_percentages = (df.isnull().sum() / len(df) * 100).to_dict()
        return missing_percentages
    
    def validate_uniqueness(self, df: pd.DataFrame, unique_columns: List[str]) -> Dict[str, int]:
        """Check for duplicate values in columns that should be unique"""
        duplicates = {}
        for col in unique_columns:
            if col in df.columns:
                duplicates[col] = df[col].duplicated().sum()
        return duplicates
    
    def validate_data_types(self, df: pd.DataFrame, expected_types: Dict[str, str]) -> Dict[str, bool]:
        """Validate data types match expectations"""
        type_validation = {}
        for col, expected_type in expected_types.items():
            if col in df.columns:
                actual_type = str(df[col].dtype)
                type_validation[col] = expected_type in actual_type
        return type_validation
    
    def validate_value_ranges(self, df: pd.DataFrame, value_ranges: Dict[str, tuple]) -> Dict[str, Dict]:
        """Check if values fall within expected ranges"""
        range_validation = {}
        for col, (min_val, max_val) in value_ranges.items():
            if col in df.columns and pd.api.types.is_numeric_dtype(df[col]):
                out_of_range = ((df[col] < min_val) | (df[col] > max_val)).sum()
                range_validation[col] = {
                    'out_of_range_count': out_of_range,
                    'percentage': (out_of_range / len(df)) * 100
                }
        return range_validation
    
    def generate_quality_report(self, df: pd.DataFrame, 
                              unique_columns: List[str] = None,
                              expected_types: Dict[str, str] = None,
                              value_ranges: Dict[str, tuple] = None) -> Dict[str, Any]:
        """Generate comprehensive data quality report"""
        report = {
            'dataset_shape': df.shape,
            'completeness': self.validate_completeness(df),
            'basic_stats': df.describe().to_dict()
        }
        
        if unique_columns:
            report['uniqueness'] = self.validate_uniqueness(df, unique_columns)
        
        if expected_types:
            report['data_types'] = self.validate_data_types(df, expected_types)
        
        if value_ranges:
            report['value_ranges'] = self.validate_value_ranges(df, value_ranges)
        
        return report

# Example usage

    df,
    unique_columns=['country', 'date'],
    expected_types={'value': 'float', 'date': 'int'},
